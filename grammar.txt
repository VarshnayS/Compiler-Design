MiniC Grammar in BNF
====================

This is the core MiniC grammar from the coursework specification.

-- Adding an extra start production 
program' ::= program EOF 

program ::= extern_list decl_list
         | decl_list

-- Direct left recursion removed
extern_list ::= extern extern_list' 
extern_list' ::= extern extern_list'|epsilon

extern ::= "extern" type_spec IDENT "(" params ")" ";"

-- Direct left recursion removed
decl_list     ::= decl decl_list'
decl_list'    ::= decl decl_list'|epsilon

-- Left factored
decl ::= "void" IDENT fun_rest | var_type IDENT decl_tail 
decl_tail ::= ";" | fun_rest 

-- improving readability of the repeated tail block 
fun_rest ::= "(" params ")" block

type_spec ::= "void" | var_type
 
var_type ::= "int" | "float" | "bool"
 
params ::= param_list | "void" | epsilon 

-- Direct left recursion removed
param_list ::= param param_list'
param_list' ::= "," param param_list' | epsilon

param ::= var_type IDENT

block ::= "{" local_decls stmt_list "}"

-- Direct left recursion removed 
local_decls ::= local_decl local_decls | epsilon

local_decl ::= var_type IDENT ";"

-- Direct left recursion removed (beta = epsilon)
stmt_list ::= stmt stmt_list | epsilon

stmt ::= expr_stmt | block | if_stmt | while_stmt | return_stmt

expr_stmt ::= expr ";" | ";"

while_stmt ::= "while" "(" expr ")" stmt

if_stmt ::= "if" "(" expr ")" block else_stmt

else_stmt ::= "else" block | epsilon

-- Left factored
return_stmt ::= "return" return_tail
return_tail ::= ";" | expr ";"

-- Not LL(1): both alternatives can start with IDENT. Resolved with LL(2):
-- IDENT followed by "=" selects assignment.
expr ::= IDENT "=" expr | rval

-- Ambiguity removed (one level per precedence, loosest first),
-- then direct left recursion removed from each level
rval ::= and_expr rval'
rval' ::= "||" and_expr rval' | epsilon

and_expr ::= eq_expr and_expr'
and_expr' ::= "&&" eq_expr and_expr' | epsilon

eq_expr ::= relational_expr eq_expr'
eq_expr' ::= "==" relational_expr eq_expr' | "!=" relational_expr eq_expr' | epsilon

relational_expr ::= add_expr relational_expr'
relational_expr' ::= "<=" add_expr relational_expr' | "<" add_expr relational_expr'
           | ">=" add_expr relational_expr' | ">" add_expr relational_expr' | epsilon

add_expr ::= mul_expr add_expr'
add_expr' ::= "+" mul_expr add_expr' | "-" mul_expr add_expr' | epsilon

mul_expr ::= unary mul_expr'
mul_expr' ::= "*" unary mul_expr' | "/" unary mul_expr' | "%" unary mul_expr' | epsilon

unary ::= "-" unary | "!" unary | primary

-- Left factored
primary ::= "(" expr ")" | IDENT identifier_tail | INT_LIT | FLOAT_LIT | BOOL_LIT
identifier_tail ::= "(" args ")" | epsilon

args ::= arg_list | epsilon

-- Direct left recursion removed
arg_list ::= expr arg_list'
arg_list' ::= "," expr arg_list' | epsilon
